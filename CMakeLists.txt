cmake_minimum_required(VERSION 3.20)
project(lob_engine CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG")
add_compile_options(-Wall -Wextra)
include_directories(include)
find_package(Threads REQUIRED)

add_executable(replay src/replay.cpp)
target_link_libraries(replay Threads::Threads)
add_executable(bench_matching bench/bench_matching.cpp)
add_executable(bench_spsc bench/bench_spsc.cpp)
target_link_libraries(bench_spsc Threads::Threads)
add_executable(unit_tests tests/test_main.cpp)
target_link_libraries(unit_tests Threads::Threads)

enable_testing()
add_test(NAME unit COMMAND unit_tests)

# libFuzzer target (clang only): cmake -DCMAKE_CXX_COMPILER=clang++ -DLOB_FUZZ=ON
option(LOB_FUZZ "build the libFuzzer harness" OFF)
if(LOB_FUZZ)
  add_executable(fuzz_itch fuzz/fuzz_itch.cpp)
  target_compile_options(fuzz_itch PRIVATE -fsanitize=fuzzer,address,undefined -g -O1)
  target_link_options(fuzz_itch PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
